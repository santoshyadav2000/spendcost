{{- if not .Values.global.database.existingDatabase }}

Database credentials:
  Secret:   {{ .Release.Name }}-postgres-auth (namespace: {{ .Values.global.namespace.name }})
  Contains: POSTGRES_USER, POSTGRES_PASSWORD, POSTGRES_DB
  kubectl get secret {{ .Release.Name }}-postgres-auth -n {{ .Values.global.namespace.name }} -o yaml

  This Secret is protected (helm.sh/resource-policy: keep) — "helm uninstall"
  will report it as kept, which is expected, not an error. This keeps the
  saved password in sync with the bundled Postgres data across mode switches
  and reinstalls (see the IMPORTANT block at the top). To also remove it for
  a full wipe:
    kubectl delete secret {{ .Release.Name }}-postgres-auth -n {{ .Values.global.namespace.name }}
{{- end }}

Backend application secrets:
  Secret:   {{ .Release.Name }}-backend-secret (namespace: {{ .Values.global.namespace.name }})

  kubectl get secret {{ .Release.Name }}-backend-secret -n {{ .Values.global.namespace.name }} -o yaml
